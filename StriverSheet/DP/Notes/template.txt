unordered_map<int,int>dp;
int fib(int n)
{
    if(n==0 || n==1)
    {
        return n;//do not touch the base case
    }
     if(dp.find(n)!=dp.end())//basic heapmap technique to find the element
     {
        return dp[n];//it will give the number bcz in hashmap index gives number
     }
     int a1=fib(n-1);
     int a2=fib(n-2);
     int ans=a1+a2;
        dp[n]=ans;//just write it in dp named hashmap i.e baisc remembering technique
     return ans;
}
